function classifyEntity(name: string, context: string) {
  if (/\b(department|agency|office|authority|city of|state of)\b/i.test(context)) {
    return { type: "GOVERNMENT", confidence: 0.95 };
  }

  if (/\b(program|fund|initiative|grant|allocation)\b/i.test(context)) {
    return { type: "PROGRAM", confidence: 0.9 };
  }

  if (/\b(inc|llc|corp|company|group)\b/i.test(name)) {
    return { type: "ORG", confidence: 0.9 };
  }

  if (/^[A-Z][a-z]+\s[A-Z][a-z]+$/.test(name)) {
    return { type: "PERSON", confidence: 0.75 };
  }

  return { type: "UNKNOWN", confidence: 0.3 };
}

function dedupeEntities(entities) {
  const map = new Map();
  for (const e of entities) {
    const key = e.name.toLowerCase();
    if (!map.has(key)) map.set(key, e);
  }
  return Array.from(map.values());
}

function extractBudgetRows(text: string) {
  const regex = /([A-Z][A-Za-z\s]+)\s+(\d{1,3}(,\d{3})+)/g;
  const results = [];
  let match;

  while ((match = regex.exec(text))) {
    const label = match[1].trim();
    const amount = match[2].replace(/,/g, "");

    results.push({
      programName: label,
      amount: Number(amount),
      amountDisplay: "$" + match[2],
      signalType: "ALLOCATION",
      confidence: 0.95
    });
  }

  return results;
}

function isValidTimelineEvent(event, docDate) {
  if (!event.date) return false;

  const eventYear = new Date(event.date).getFullYear();
  const docYear = new Date(docDate).getFullYear();

  if (Math.abs(eventYear - docYear) > 2) return false;

  if (
    /\b(episode|season|recap|watch|documentary)\b/i.test(event.text)
  ) return false;

  return true;
}
